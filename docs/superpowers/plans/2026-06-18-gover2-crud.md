# gover2-crud Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add full Create / Read / Update / Delete for all 7 asset categories to the gover2 Wails app.

**Architecture:** Extend the existing read-only layer in-place: new Input structs in models, write functions in services, bridge methods on `*App`, typed wrappers in api/index.ts, a real EditPanel.vue, a new ConfirmDialog.vue, and CRUD wiring in all 7 views. DataTable gains `row-click` and `delete` emits.

**Tech Stack:** Go 1.21, modernc.org/sqlite (in-memory for tests), Wails v2, Vue 3 + TypeScript, Vitest + @vue/test-utils, Tailwind CSS, pnpm.

## Global Constraints

- Working directory for all Go commands: `/home/mg/inventory/gover2`
- Working directory for all frontend commands: `/home/mg/inventory/gover2/frontend`
- Do NOT add new packages or architectural layers — extend existing files only, except where "Create" is explicit.
- `updated_at` is never included in Input structs — always set by SQL `datetime('now','localtime')`.
- All bridge methods must open with `if a.db == nil { return errNoDB }` (or `return nil, errNoDB` for slice returns).
- After a successful write the frontend calls the relevant `List*` method to refresh — the bridge does not push events.
- Test coverage targets: `internal/services` ≥ 70%, `internal/bridge` ≥ 50%.
- Frontend tests run with `pnpm run test` from `/home/mg/inventory/gover2/frontend`.
- Go tests run with `go test ./...` from `/home/mg/inventory/gover2`.

```go
func InsertComputer(db *sql.DB, d models.ComputerInput) (int64, error) {
	res, err := db.Exec(
		`INSERT INTO computers (name, model, user_id, status, purchase_date, warranty_expiry, notes)
		 VALUES (?, ?, ?, ?, ?, ?, ?)`,
		d.Name, d.Model, d.UserID, d.Status, d.PurchaseDate, d.WarrantyExpiry, d.Notes,
	)
	if err != nil {
		return 0, err
	}
	return res.LastInsertId()
}

func UpdateComputer(db *sql.DB, id int, d models.ComputerInput) error {
	_, err := db.Exec(
		`UPDATE computers SET name=?, model=?, user_id=?, status=?, purchase_date=?,
		 warranty_expiry=?, notes=?, updated_at=datetime('now','localtime') WHERE id=?`,
		d.Name, d.Model, d.UserID, d.Status, d.PurchaseDate, d.WarrantyExpiry, d.Notes, id,
	)
	return err
}

func DeleteComputer(db *sql.DB, id int) error {
	_, err := db.Exec(`DELETE FROM computers WHERE id=?`, id)
	return err
}
```

```go
package services

import (
	"database/sql"

	"gover2/internal/models"
)

func ListUsersForDropdown(db *sql.DB) ([]models.DropdownItem, error) {
	rows, err := db.Query(`SELECT id, COALESCE(name,'') FROM users ORDER BY name`)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []models.DropdownItem
	for rows.Next() {
		var item models.DropdownItem
		if err := rows.Scan(&item.ID, &item.Name); err != nil {
			return nil, err
		}
		out = append(out, item)
	}
	return out, rows.Err()
}

func ListDevicesForDropdown(db *sql.DB) ([]models.DeviceDropdownItem, error) {
	rows, err := db.Query(`
		SELECT id, COALESCE(name,''), 'computer' AS kind FROM computers
		UNION ALL SELECT id, COALESCE(name,''), 'smartphone' FROM smartphones
		UNION ALL SELECT id, COALESCE(name,''), 'tablet'     FROM tablets
		ORDER BY kind, name`)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []models.DeviceDropdownItem
	for rows.Next() {
		var item models.DeviceDropdownItem
		if err := rows.Scan(&item.ID, &item.Name, &item.Kind); err != nil {
			return nil, err
		}
		out = append(out, item)
	}
	return out, rows.Err()
}
```
